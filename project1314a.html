"""01. 급식 잔반 개선 대상 찾기
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from math import isfinite

def find_menus(rows, threshold=0.30):
    if not isfinite(threshold) or not 0 <= threshold <= 1:
        raise ValueError("기준은 0~1의 유한값이어야 합니다")
    found = []
    for name, served, waste in rows:
        if not all(isfinite(x) for x in (served, waste)):
            raise ValueError("양은 유한값이어야 합니다")
        if served <= 0 or not 0 <= waste <= served:
            raise ValueError("제공량·잔반량 범위를 확인하세요")
        rate = waste / served
        if rate >= threshold:
            found.append((name, round(rate, 4)))
    return found

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    assert find_menus([("국", 100, 30), ("밥", 100, 10)]) == [("국", 0.3)]
    assert find_menus([]) == []
    must_raise(ValueError, lambda: find_menus([("오류", 0, 0)]))
    print(find_menus([("채소반찬", 1000, 350), ("밥", 2000, 300), ("국", 1000, 300)]))
    print("정상·경계·예외 테스트 통과")

"""02. 교실 소음 주의 구간 찾기
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.

from math import isfinite

def noisy_runs(values, threshold=70, min_run=2):
    if not isinstance(min_run, int) or min_run < 1:
        raise ValueError("최소 길이는 양의 정수여야 합니다")
    if not isfinite(threshold) or any(not isfinite(v) or v < 0 for v in values):
        raise ValueError("측정값과 기준을 확인하세요")
    result, start = [], None
    for i in range(len(values) + 1):
        high = i < len(values) and values[i] >= threshold
        if high and start is None:
            start = i
        if not high and start is not None:
            if i - start >= min_run:
                result.append((start, i - 1))
            start = None
    return result

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    assert noisy_runs([65, 70, 72, 60, 80, 80]) == [(1, 2), (4, 5)]
    assert noisy_runs([]) == []
    assert noisy_runs([70, 70]) == [(0, 1)]
    must_raise(ValueError, lambda: noisy_runs([60], min_run=0))
    print(noisy_runs([65, 70, 72, 60, 80, 80]))
    print("정상·경계·예외 테스트 통과")

"""03. 통학·공공교통 최소 시간 경로
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from heapq import heappush, heappop
from math import inf, isfinite

def shortest_time(graph, start, goal):
    if start not in graph or goal not in graph:
        raise ValueError("출발·도착 정점이 필요합니다")
    for edges in graph.values():
        for node, weight in edges:
            if node not in graph or not isfinite(weight) or weight < 0:
                raise ValueError("연결 정점과 음수 아닌 비용을 확인하세요")
    dist, prev = {start: 0}, {}
    queue = [(0, start)]
    while queue:
        cost, node = heappop(queue)
        if cost != dist.get(node, inf):
            continue
        if node == goal:
            path = [goal]
            while path[-1] != start:
                path.append(prev[path[-1]])
            return cost, path[::-1]
        for nxt, weight in graph[node]:
            new = cost + weight
            if new < dist.get(nxt, inf):
                dist[nxt], prev[nxt] = new, node
                heappush(queue, (new, nxt))
    return None, []

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    g = {"집": [("환승", 4), ("학교", 12)], "환승": [("학교", 3)], "학교": [], "외곽": []}
    assert shortest_time(g, "집", "학교") == (7, ["집", "환승", "학교"])
    assert shortest_time(g, "집", "집") == (0, ["집"])
    assert shortest_time(g, "집", "외곽") == (None, [])
    must_raise(ValueError, lambda: shortest_time({"A": [("B", -1)], "B": []}, "A", "B"))
    print(shortest_time(g, "집", "학교"))
    print("정상·경계·예외 테스트 통과")

계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.

def best_bundle(items, budget):
    if not isinstance(budget, int) or budget < 0:
        raise ValueError("예산은 음수 아닌 정수여야 합니다")
    if any(not isinstance(p, int) or not isinstance(v, int) or p < 0 or v < 0
           for _, p, v in items):
        raise ValueError("가격·점수는 음수 아닌 정수여야 합니다")
    best_score, best_cost, best_ids = -1, 0, []
    chosen = []

    def visit(i, cost, score):
        nonlocal best_score, best_cost, best_ids
        if cost > budget:
            return
        if i == len(items):
            if score > best_score:
                best_score, best_cost, best_ids = score, cost, chosen[:]
            return
        visit(i + 1, cost, score)
        chosen.append(i)
        _, price, value = items[i]
        visit(i + 1, cost + price, score + value)
        chosen.pop()

    visit(0, 0, 0)
    return best_score, best_cost, [items[i][0] for i in best_ids]

def first_index(numbers, target):
    if any(a > b for a, b in zip(numbers, numbers[1:])):
        raise ValueError("목록이 오름차순이어야 합니다")
    lo, hi = 0, len(numbers)

        mid = (lo + hi) // 2
        if numbers[mid] < target:
            lo = mid + 1
        else:
            hi = mid
    return lo if lo < len(numbers) and numbers[lo] == target else -1

def minimum_radius(positions, k):
    if not isinstance(k, int) or k < 1 or any(not isinstance(x, int) for x in positions):
        raise ValueError("정수 위치와 양의 정수 거점 수가 필요합니다")
    xs = sorted(positions)
    if not xs:
        return 0

    def feasible(radius):
        count, i = 0, 0
        while i < len(xs):
            limit = xs[i] + 2 * radius
            count += 1
            while i < len(xs) and xs[i] <= limit:
                i += 1
        return count <= k

    lo, hi = 0, xs[-1] - xs[0]
    while lo < hi:
        mid = (lo + hi) // 2
        if feasible(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo

"""07. 침수 격자에서 대피 경로 찾기
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from collections import deque

def evacuation_steps(grid, start, goal):
    if not grid or not grid[0] or any(len(row) != len(grid[0]) for row in grid):
        raise ValueError("비어 있지 않은 직사각형 격자가 필요합니다")
    if any(v not in (0, 1) for row in grid for v in row):
        raise ValueError("격자 값은 0 또는 1이어야 합니다")
    h, w = len(grid), len(grid[0])
    for r, c in (start, goal):
        if not (0 <= r < h and 0 <= c < w) or grid[r][c] != 0:
            raise ValueError("시작·목표는 통행 가능한 격자 안이어야 합니다")
    queue, seen = deque([(start, 0)]), {start}
    while queue:
        (r, c), distance = queue.popleft()
        if (r, c) == goal:
            return distance
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < h and 0 <= nc < w and grid[nr][nc] == 0 and (nr, nc) not in seen:
                seen.add((nr, nc))
                queue.append(((nr, nc), distance + 1))
    return None

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from itertools import combinations

def minimum_hubs(needed, hubs):
    needed = set(needed)
    if any(not isinstance(areas, (set, frozenset)) for areas in hubs.values()):
        raise ValueError("담당 지역은 집합으로 입력하세요")
    names = sorted(hubs)
    for size in range(len(names) + 1):
        for choice in combinations(names, size):
            covered = set()
            for name in choice:

def disconnected(graph, start):
    if start not in graph:
        raise ValueError("기준 측정소가 없습니다")
    adj = {node: set(neighbors) for node, neighbors in graph.items()}
    for node, neighbors in adj.items():
        for nxt in neighbors:
            if nxt not in adj or node not in adj[nxt]:
                raise ValueError("무방향 연결을 양쪽에 명시하세요")
    seen, stack = {start}, [start]
    while stack:
        node = stack.pop()
        for nxt in adj[node]:
            if nxt not in seen:
                seen.add(nxt)
                stack.append(nxt)
    return sorted(seen), sorted(set(graph) - seen)

        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

"""11. 가상 접촉망의 연결 단계 탐색
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from collections import deque

def connection_levels(graph, start):
    if start not in graph:
        raise ValueError("시작 정점이 없습니다")
    adj = {node: set(ns) for node, ns in graph.items()}
    for node, ns in adj.items():
        for nxt in ns:
            if nxt not in adj or node not in adj[nxt]:
                raise ValueError("무방향 관계를 확인하세요")
    distance, queue = {start: 0}, deque([start])
    while queue:
        node = queue.popleft()
        for nxt in graph[node]:
            if nxt not in distance:
                distance[nxt] = distance[node] + 1
                queue.append(nxt)
    return distance

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    g = {"A": ["B", "C"], "B": ["A", "D"], "C": ["A"], "D": ["B"], "E": []}
    assert connection_levels(g, "A") == {"A": 0, "B": 1, "C": 1, "D": 2}
    assert connection_levels({"A": []}, "A") == {"A": 0}
    must_raise(ValueError, lambda: connection_levels(g, "Z"))
    print(connection_levels(g, "A"))
    print("정상·경계·예외 테스트 통과")

"""12. 도로 통제 구역 우회 탐색
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from heapq import heappush, heappop
from math import inf

def astar_steps(grid, start, goal):
    if not grid or not grid[0] or any(len(row) != len(grid[0]) for row in grid):
        raise ValueError("직사각형 격자가 필요합니다")
    if any(v not in (0, 1) for row in grid for v in row):
        raise ValueError("격자 값은 0 또는 1이어야 합니다")
    h, w = len(grid), len(grid[0])
    for r, c in (start, goal):
        if not (0 <= r < h and 0 <= c < w) or grid[r][c]:
            raise ValueError("시작·목표는 통행 가능해야 합니다")

    def heuristic(pos):
        return abs(pos[0] - goal[0]) + abs(pos[1] - goal[1])

    best, queue = {start: 0}, [(heuristic(start), 0, start)]
    while queue:
        _, cost, node = heappop(queue)
        if cost != best.get(node, inf):
            continue
        if node == goal:
            return cost
        r, c = node
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nxt = (r + dr, c + dc)
            nr, nc = nxt
            if 0 <= nr < h and 0 <= nc < w and grid[nr][nc] == 0:
                new = cost + 1
                if new < best.get(nxt, inf):
                    best[nxt] = new
                    heappush(queue, (new + heuristic(nxt), new, nxt))
    return None

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    grid = [[0, 0, 1], [1, 0, 0], [0, 0, 0]]

    assert astar_steps([[0]], (0, 0), (0, 0)) == 0
    assert astar_steps([[0, 1, 0]], (0, 0), (0, 2)) is None
    must_raise(ValueError, lambda: astar_steps([[0, 2]], (0, 0), (0, 1)))
    print(astar_steps(grid, (0, 0), (2, 2)))

"""13. 재난 안내문의 핵심 표현 찾기
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
def kmp_positions(text, pattern):
    if not pattern:
        raise ValueError("검색어는 비어 있을 수 없습니다")
    lps, j = [0] * len(pattern), 0
    for i in range(1, len(pattern)):
        while j and pattern[i] != pattern[j]:
            j = lps[j - 1]
        if pattern[i] == pattern[j]:
            j += 1
            lps[i] = j
    result, j = [], 0
    for i, char in enumerate(text):
        while j and char != pattern[j]:
            j = lps[j - 1]
        if char == pattern[j]:
            j += 1
        if j == len(pattern):
            result.append(i - len(pattern) + 1)
            j = lps[j - 1]
    return result

"""
from bisect import bisect_left, bisect_right

def archive_range(records, start, end):
    if not isinstance(start, int) or not isinstance(end, int) or start > end:
        raise ValueError("정수 연도 범위를 확인하세요")
    if any(not isinstance(year, int) for year, _ in records):
        raise ValueError("연도는 정수여야 합니다")
    ordered = sorted(records, key=lambda row: row[0])
    years = [year for year, _ in ordered]
    return ordered[bisect_left(years, start):bisect_right(years, end)]

def must_raise(error_type, function):
    try:

    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

END = ""  # 문자를 키로 쓰므로 빈 문자열은 종료 표시로 사용 가능

def build_trie(words):
    root = {}
    for word in words:
        if not isinstance(word, str) or not word:
            raise ValueError("비어 있지 않은 문자열 용어가 필요합니다")
        node = root
        for char in word:
            node = node.setdefault(char, {})
        node[END] = True
    return root

def prefix_search(root, prefix):
    node = root
    for char in prefix:
        if char not in node:
            return []
        node = node[char]
    found, stack = [], [(node, prefix)]
    while stack:
        current, word = stack.pop()
        if END in current:
            found.append(word)
        for char, child in current.items():
            if char != END:
                stack.append((child, word + char))
    return sorted(found)

        function()
    except error_type:
        return

def feasible_allocations(minimums, total):
    if len(minimums) != 3:
        raise ValueError("이 예제는 세 집단을 다룹니다")
    if not isinstance(total, int) or total < 0 or any(not isinstance(x, int) or x < 0 for x in minimums):
        raise ValueError("음수 아닌 정수 자원이 필요합니다")
    a_min, b_min, c_min = minimums
    answers = []
    for a in range(a_min, total + 1):
        for b in range(b_min, total - a + 1):
            c = total - a - b
            if c >= c_min:
                answers.append((a, b, c))
    return answers

        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.

from math import isfinite

def pareto_candidates(scores):
    if not scores:
        return []
    size = len(next(iter(scores.values())))
    if size == 0 or any(len(v) != size or any(not isfinite(x) for x in v) for v in scores.values()):
        raise ValueError("같은 개수의 유한 점수가 필요합니다")
    result = []
    for name, a in scores.items():
        dominated = False
        for other, b in scores.items():
            if other != name and all(y >= x for x, y in zip(a, b)) and any(y > x for x, y in zip(a, b)):
                dominated = True
                break
        if not dominated:
            result.append(name)
    return result

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":

    assert pareto_candidates({"A": (1, 1), "B": (1, 1)}) == ["A", "B"]
    assert pareto_candidates({}) == []
    must_raise(ValueError, lambda: pareto_candidates({"A": (1,), "B": (1, 2)}))
    print(pareto_candidates(scores))
    print("정상·경계·예외 테스트 통과")

계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.

from itertools import product

def counterexamples(form):
    if form not in {"MP", "AC"}:
        raise ValueError("MP 또는 AC 형식을 선택하세요")
    result = []
    for p, q in product((False, True), repeat=2):
        implication = (not p) or q
        premises = (implication, p) if form == "MP" else (implication, q)
        conclusion = q if form == "MP" else p
        if all(premises) and not conclusion:
            result.append({"P": p, "Q": q})
    return result

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    assert counterexamples("MP") == []
    assert counterexamples("AC") == [{"P": False, "Q": True}]
    must_raise(ValueError, lambda: counterexamples("UNKNOWN"))
    print("MP 반례:", counterexamples("MP"))
    print("AC 반례:", counterexamples("AC"))
    print("정상·경계·예외 테스트 통과")

"""20. 주장에서 원출처까지 경로 찾기
수업용 가상 자료. Python 3.10 이상 / 표준 라이브러리만 사용.
계산 결과를 실제 정책·진단·가치판단으로 간주하지 않습니다.
"""
from collections import deque

def source_path(graph, start, originals):
    originals = set(originals)
    if start not in graph or not originals <= set(graph):
        raise ValueError("시작과 원자료 후보가 그래프 안에 있어야 합니다")
    if any(nxt not in graph for ns in graph.values() for nxt in ns):
        raise ValueError("인용 대상 문서가 누락되었습니다")
    queue, parent = deque([start]), {start: None}
    while queue:
        node = queue.popleft()
        if node in originals:
            path = []
            while node is not None:
                path.append(node)
                node = parent[node]
            return path[::-1]
        for nxt in graph[node]:
            if nxt not in parent:
                parent[nxt] = node
                queue.append(nxt)
    return None

def must_raise(error_type, function):
    try:
        function()
    except error_type:
        return
    raise AssertionError(f"{error_type.__name__} 예외가 발생해야 합니다")

if __name__ == "__main__":
    g = {"게시글": ["기사"], "기사": ["보고서", "게시글"], "보고서": []}
    assert source_path(g, "게시글", {"보고서"}) == ["게시글", "기사", "보고서"]
    assert source_path(g, "보고서", {"보고서"}) == ["보고서"]
    assert source_path({"A": ["B"], "B": ["A"], "C": []}, "A", {"C"}) is None
    must_raise(ValueError, lambda: source_path(g, "게시글", {"없는문서"}))
    print(source_path(g, "게시글", {"보고서"}))
    print("정상·경계·예외 테스트 통과")
