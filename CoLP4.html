프로젝트 1 · 프로그래밍 — 재귀·반복 계산 협업 라이브러리

팩토리얼을 두 방식으로 계산하고 결과를 비교한다.

성공 조건: 모듈별 단위 시험 후 통합 출력이 예시와 일치하고, 정상·경계·예외 입력을 설명할 수 있어야 합니다.

학생 1 · 재귀 담당 — recursive_calc.py
def 팩토리얼_재귀(n):
    """0 이상의 정수 n의 팩토리얼"""
    if not isinstance(n, int) or n < 0:
        raise ValueError("0 이상의 정수가 필요합니다")
    if n <= 1:
        return 1
    return n * 팩토리얼_재귀(n - 1)
학생 2 · 반복 담당 — iterative_calc.py
def 팩토리얼_반복(n):
    if not isinstance(n, int) or n < 0:
        raise ValueError("0 이상의 정수가 필요합니다")
    결과 = 1
    for 수 in range(2, n + 1):
        결과 *= 수
    return 결과
학생 3 · 검증 담당 — compare.py
from recursive_calc import 팩토리얼_재귀
from iterative_calc import 팩토리얼_반복

def 비교(n):
    재귀값 = 팩토리얼_재귀(n)
    반복값 = 팩토리얼_반복(n)
    assert 재귀값 == 반복값
    return 재귀값, 반복값
학생 4 · 보고 담당 — report.py
def 보고(n, 결과):
    재귀값, 반복값 = 결과
    return f"{n}! = {재귀값}, 반복 결과 = {반복값}"
공동 통합 — main.py
from compare import 비교
from report import 보고

학생 1 · 큐 담당 — queue_service.py
from collections import deque

class 대기열:
    def __init__(self):
        self.자료 = deque()
    def 추가(self, 이름):
        self.자료.append(이름)
    def 처리(self):
        if not self.자료:
            raise IndexError("대기열이 비었습니다")
        return self.자료.popleft()
학생 2 · 스택 담당 — stack_history.py
class 반납기록:
    def __init__(self):
        self.자료 = []
    def 기록(self, 책):
        self.자료.append(책)
    def 최근(self):
        if not self.자료:
            raise IndexError("기록이 비었습니다")
        return self.자료[-1]

학생 2 · 탐색 담당 — search_bfs.py
from collections import deque

def 최단경로(그래프, 시작, 목표):
    if 시작 not in 그래프 or 목표 not in 그래프:
        raise ValueError("존재하지 않는 정점")
    대기 = deque([(시작, [시작])])
    방문 = {시작}
    while 대기:
        현재, 경로 = 대기.popleft()
        if 현재 == 목표:
            return 경로
        for 다음 in 그래프[현재]:
            if 다음 not in 방문:
                방문.add(다음)
                대기.append((다음, 경로 + [다음]))
    return None
학생 3 · 검증 담당 — validate.py
def 경로검증(그래프, 경로):
    if not 경로:
        return False
    return all(b in 그래프[a] for a, b in zip(경로, 경로[1:]))

def 사용전력(기기):
    """기기: (와트, 시간) 쌍의 목록, 반환 단위 Wh"""
    if any(w < 0 or h < 0 for w, h in 기기):
        raise ValueError("음수 전력 또는 시간")
    return sum(w * h for w, h in 기기)
학생 2 · 절감 계산 담당 — saving.py
def 절감후(사용량, 절감률):
    if not 0 <= 절감률 <= 1:
        raise ValueError("절감률은 0~1")
    return 사용량 * (1 - 절감률)
학생 3 · 평가 담당 — evaluation.py
def 평가(기존, 개선):
    if 기존 < 0 or 개선 < 0:
        raise ValueError("사용량은 음수가 될 수 없습니다")
    return "절감 성공" if 개선 < 기존 else "개선 필요"
