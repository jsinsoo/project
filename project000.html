    rate = waste / served * 100
    results.append((name, round(rate, 1)))

def is_prime(n):
    if n < 2: return False, 0
    checks = 0
    for d in range(2, int(n ** 0.5) + 1):
        checks += 1
        if n % d == 0: return False, checks
    return True, checks
for n in [1, 2, 49, 97, 101]:
    print(n, is_prime(n))

memo = {0: 0, 1: 1}
computed = 0
def fib(n):
    global computed
    if n not in memo:
        computed += 1
        memo[n] = fib(n-1) + fib(n-2)
    return memo[n]

grid = ["...#", ".#..", "...."]
start, end = (0, 0), (2, 3)
q = deque([(start, 0)])
seen = {start}
while q:
    (r, c), dist = q.popleft()
    if (r, c) == end:
        print("최단 거리:", dist)
        break
    for dr, dc in [(1,0),(-1,0),(0,1),(0,-1)]:
        nr, nc = r+dr, c+dc
        if 0 <= nr < len(grid) and 0 <= nc < len(grid[0]) and grid[nr][nc] == "." and (nr,nc) not in seen:
            seen.add((nr,nc)); q.append(((nr,nc),dist+1))
else:
    print("도달 불가")

for p, q in product([False, True], repeat=2):
    premise1 = (not p) or q  # P이면 Q
    premise2 = p
    conclusion = q
    if premise1 and premise2 and not conclusion:
        counterexamples.append((p, q))

print("타당" if not counterexamples else "타당하지 않음")
